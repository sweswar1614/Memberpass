const registerUser = async (req, res) => {
  try {
    const { name, email, password, phone } = req.body;

    if (!name || !email || !password) {
      return res.status(400).json({
        success: false,
        message: "Name, email and password are required"
      });
    }

    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
    if (!emailRegex.test(email.trim())) {
      return res.status(400).json({
        success: false,
        message: "Please provide a valid email address"
      });
    }

    if (password.length < 6) {
      return res.status(400).json({
        success: false,
        message: "Password must contain at least 6 characters"
      });
    }

    const normalizedEmail = email.trim().toLowerCase();

    const existingUser = await User.findOne({
      email: normalizedEmail
    });

    if (existingUser) {
      return res.status(409).json({
        success: false,
        message: "Email already registered"
      });
    }

    const user = await User.create({
      name: name.trim(),
      email: normalizedEmail,
      password,
      phone,
      role: "member"
    });

    const token = generateToken(user);

    return res.status(201).json({
      success: true,
      message: "Registration successful",
      token,
      user: {
        id: user._id,
        name: user.name,
        email: user.email,
        passCode: user.passCode,
        role: user.role
      }
    });
  } catch (error) {
    console.error("Registration error:", error.message);

    if (error.code === 11000) {
      return res.status(409).json({
        success: false,
        message: "Email or pass code already exists"
      });
    }

    return res.status(500).json({
      success: false,
      message: "Registration failed"
    });
  }
};

        message: "Member not found"
      });

      .populate("plan")
      .sort({ createdAt: -1 });

    // Determine current status
    let membershipStatus = "none";
    let isExpired = false;

    if (membership) {
      membershipStatus = membership.status;
      if (new Date(membership.endDate) < new Date() && membership.status === "active") {
        isExpired = true;
        membershipStatus = "ended";
      }
    }

    // Get recent payments
    const payments = await Payment.find({ user: user._id })
      .populate("plan", "name priceInPaise")
      .sort({ createdAt: -1 })
      .limit(5);

    return res.status(200).json({
      success: true,
      member: {
        id: user._id,
        name: user.name,
        email: user.email,
        phone: user.phone || "",
        role: user.role,
        passCode: user.passCode,
        membershipStatus,
        isExpired,
        membership: membership

              id: membership._id,
              plan: membership.plan
                ? {
                    id: membership.plan._id,

const createOrder = async (req, res) => {
  try {

    if (!planId && !planName) {

    if (planId && !mongoose.Types.ObjectId.isValid(planId)) {

      });
    }

    const query = planId

      amount: plan.priceInPaise,

        userEmail: req.user.email,

// @desc Verify Razorpay Payment and Activate Membership
// @route POST /api/payments/verify
// @access Private
const verifyPayment = async (req, res) => {
  try {
    const {
      paymentRecordId,
      razorpay_payment_id,
      razorpay_signature
    } = req.body;

    if (
      !paymentRecordId ||
      !razorpay_payment_id ||
      !razorpay_signature
    ) {
      return res.status(400).json({
        success: false,
        message: "Missing payment verification parameters"
      });
    }

    if (!mongoose.Types.ObjectId.isValid(paymentRecordId)) {
      return res.status(400).json({
        success: false,
        message: "Invalid paymentRecordId"
      });
    }

    const keySecret = process.env.RAZORPAY_KEY_SECRET;

    if (!keySecret?.trim()) {
      return res.status(500).json({
        success: false,
        code: "RAZORPAY_SECRET_MISSING",
        message: "Razorpay secret is not configured"
      });
    }

    // Find payment belonging to authenticated user
    const payment = await Payment.findOne({
      _id: paymentRecordId,
      user: req.user._id
    });

    if (!payment) {
      return res.status(404).json({
        success: false,
        message: "Payment record not found"
      });
    }

    // Idempotency: do not create membership twice
    if (payment.status === "captured") {
      const existingMembership = payment.membership
        ? await Membership.findById(payment.membership).populate("plan")
        : null;

      if (existingMembership) {
        return res.status(200).json({
          success: true,
          message: "Payment already verified",
          membership: {
            id: existingMembership._id,
            plan: existingMembership.plan?.name,
            status: existingMembership.status,
            startDate: existingMembership.startDate,
            endDate: existingMembership.endDate,
            classesUsed: existingMembership.classesUsed
          }
        });
      }

      return res.status(409).json({
        success: false,
        message: "Payment captured but membership record is missing. Contact support."
      });
    }

    if (payment.status !== "created") {
      return res.status(400).json({
        success: false,
        message: `Payment status is '${payment.status}'. Verification is not allowed.`
      });
    }

    const savedOrderId = payment.razorpayOrderId;

    if (!savedOrderId) {
      return res.status(500).json({
        success: false,
        message: "Saved Razorpay order ID is missing"
      });
    }

    // Verify signature using database order ID
    const expectedSignature = crypto
      .createHmac("sha256", keySecret.trim())
      .update(`${savedOrderId}|${razorpay_payment_id}`)
      .digest("hex");

    const expectedBuffer = Buffer.from(expectedSignature, "hex");
    const receivedBuffer = Buffer.from(razorpay_signature, "hex");

    if (
      expectedBuffer.length !== receivedBuffer.length ||
      !crypto.timingSafeEqual(expectedBuffer, receivedBuffer)
    ) {
      payment.status = "failed";
      await payment.save();

      return res.status(400).json({
        success: false,
        message: "Invalid payment signature"
      });
    }

    const plan = await Plan.findById(payment.plan);

    if (!plan) {
      return res.status(404).json({
        success: false,
        message: "Associated membership plan not found"
      });
    }

    if (!Number.isFinite(plan.durationDays) || plan.durationDays <= 0) {
      return res.status(400).json({
        success: false,
        message: "Invalid membership plan duration"
      });
    }

    const now = new Date();
    const endDate = new Date(
      now.getTime() + plan.durationDays * 24 * 60 * 60 * 1000
    );

    // Create membership and link it to payment.
    // Save payment as captured only after membership creation succeeds.
    const membership = await Membership.create({
      user: req.user._id,
      plan: plan._id,
      status: "active",
      startDate: now,
      endDate,
      classesUsed: 0,
      payment: payment._id
    });

    try {
      payment.status = "captured";
      payment.razorpayPaymentId = razorpay_payment_id;
      payment.razorpaySignature = razorpay_signature;
      payment.membership = membership._id;

      await payment.save();

      // End previous active memberships only after successful save
      await Membership.updateMany(
        {
          user: req.user._id,
          status: "active",
          _id: { $ne: membership._id }
        },
        {
          $set: { status: "ended" }
        }
      );

    } catch (saveError) {
      // Avoid leaving an unlinked membership if payment save fails
      await Membership.findByIdAndDelete(membership._id);
      throw saveError;
    }

    return res.status(200).json({
      success: true,
      message: "Payment verified and membership activated",
      payment: {
        id: payment._id,
        orderId: payment.razorpayOrderId,
        paymentId: payment.razorpayPaymentId,
        amount: payment.amount,
        status: payment.status
      },
      membership: {
        id: membership._id,
        plan: plan.name,
        status: membership.status,
        startDate: membership.startDate,
        endDate: membership.endDate,
        classesUsed: membership.classesUsed
      }
    });

  } catch (error) {
    console.error("verifyPayment error:", error);

    return res.status(500).json({
      success: false,
      message: "Server error verifying payment"
    });
  }
};

        }
      });
    }

        success: false,
        entryStatus: "ENTRY_DENIED",
        reason: result.error
      });
    }

    const classLimit = membership.plan?.classLimit;

    // Atomic increment with current membership conditions
    const filter = {
      _id: membership._id,
      status: "active",
      endDate: { $gte: new Date() },
      classesUsed: membership.classesUsed
    };

    if (classLimit !== null && classLimit !== undefined) {
      filter.classesUsed = {
        $eq: membership.classesUsed,
        $lt: classLimit
      };
    }

    const updatedMembership = await Membership.findOneAndUpdate(
      filter,
      { $inc: { classesUsed: 1 } },
      { new: true }
    );

    if (!updatedMembership) {
      return res.status(409).json({
        success: false,
        entryStatus: "ENTRY_DENIED",
        reason: "Membership changed or class limit reached. Please scan again."
      });
    }

    incrementedMembershipId = updatedMembership._id;
    incrementedTo = updatedMembership.classesUsed;

    // Save attendance record
    const attendance = await Attendance.create({
      member: member._id,
      membership: membership._id,
      staff: req.user._id,
      checkInId,
      checkedInAt: new Date()
    });

        plan: membership.plan?.name || "Member"
      },
      attendance: {
        id: attendance._id,
        checkedInAt: attendance.checkedInAt
      },
      classesUsed: updatedMembership.classesUsed,
      classLimit: classLimit ?? null
    });
  } catch (error) {
    // Roll back increment if attendance record was not saved
    if (incrementedMembershipId && incrementedTo !== null) {
      await Membership.updateOne(

const jwt = require("jsonwebtoken");
const User = require("../models/User");

const protect = async (req, res, next) => {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith("Bearer ")) {
      return res.status(401).json({
        success: false,
        message: "Authentication required"
      });
    }

    const token = authHeader.split(" ")[1];
    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    const user = await User.findById(decoded.id).select("-password");

    if (!user || !user.isActive) {
      return res.status(401).json({
        success: false,
        message: "Invalid or inactive account"
      });
    }

    req.user = user;
    next();
  } catch (error) {
    return res.status(401).json({
      success: false,
      message: "Invalid or expired token"
    });
  }
};

module.exports = { protect };

    const { question } = req.body;

    if (typeof question !== "string" || !question.trim()) {
      return res.status(400).json({
        success: false,
        message: "Please provide a valid question"
      });
    }

    const response = await fetch(`${AI_SERVICE_URL}/chat`, {
      method: "POST",
      headers: {
        "Content-Type": "application/json"
      },
      body: JSON.stringify({
        question: question.trim()
      }),
      signal: AbortSignal.timeout(15000)
    });

    const data = await response.json();

    if (!response.ok) {
      return res.status(response.status).json(data);
    }

    return res.json(data);
  } catch (error) {
    console.error("AI service error:", error.message);

    return res.status(503).json({
      success: false,
      message: "AI service is currently unavailable"
    });
  }
});
